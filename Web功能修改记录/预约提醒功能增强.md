### 2. UltimateFix规则管理增强

#### 核心逻辑重构
```typescript
// UltimateFix类增强
class UltimateFix {
  private dialogProvider?: DialogContextType;
  
  constructor(dialogProvider?: DialogContextType) {
    this.dialogProvider = dialogProvider;
  }
  
  // 智能规则选择逻辑
  private async selectRule(rules: Rule[]): Promise<Rule | null> {
    if (this.dialogProvider) {
      // 使用自定义对话框
      const ruleNames = rules.map((rule, index) => `${index + 1}. ${rule.name}`).join('\n');
      const selection = await this.dialogProvider.showPrompt(
        '选择规则',
        `请选择要使用的规则:\n${ruleNames}\n\n请输入规则编号:`,
        '1'
      );
      // 处理选择逻辑
    } else {
      // 降级到原生prompt
      const ruleNames = rules.map((rule, index) => `${index + 1}. ${rule.name}`).join('\n');
      const selection = prompt(`请选择要使用的规则:\n${ruleNames}\n\n请输入规则编号:`);
      // 处理选择逻辑
    }
  }
  
  // 规则创建增强
  private async createNewRule(): Promise<Rule | null> {
    const ruleName = await this.promptForRuleName();
    if (!ruleName) return null;
    
    const ruleType = await this.promptForRuleType();
    if (!ruleType) return null;
    
    // 创建规则逻辑
    return this.buildRule(ruleName, ruleType);
  }
}
```
